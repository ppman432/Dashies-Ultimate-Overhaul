scripted_diplomatic_actions = {

    break_ceasefire = {
        # ROOT = sender, current scope = receiver (target)

        icon = 1

        requires_acceptance = no

        visible = {
            # Check if the target has a ceasefire with us (ROOT)
            has_country_flag = ceasefire_active_with_@ROOT
            # Restrict this action to the two ceasefire signatories (initiator and receiver), not wider war participants.
            OR = {
                has_country_flag = ceasefire_initiated_by_@ROOT
                has_country_flag = ceasefire_initiator_with_@ROOT
            }
            NOT = { has_war_with = ROOT }
            
            # Only show if there's an active truce or NAP
            # OR = {
            #     has_non_aggression_pact_with = ROOT
            #     has_truce_with = ROOT
            # }
        }

        selectable = {
            # Game rule: optionally block AI from breaking ceasefires.
            OR = {
                ROOT = { is_ai = no }
                has_game_rule = { rule = ceasefire_ai_break_rule option = ALLOWED }
            }

            custom_trigger_tooltip = {
                tooltip = ceasefire_break_requires_ceasefire
                has_country_flag = ceasefire_active_with_@ROOT
            }
            custom_trigger_tooltip = {
                tooltip = ceasefire_break_only_initiator_tt
                OR = {
                    has_country_flag = ceasefire_initiated_by_@ROOT
                    has_country_flag = ceasefire_initiator_with_@ROOT
                }
            }
            custom_trigger_tooltip = {
                tooltip = ceasefire_break_available_tt
                # Cooldown: use "days since flag was set" (reliable) instead of relying on timed flag expiry.
                OR = {
                    NOT = { has_country_flag = ceasefire_break_cooldown }
                    AND = {
                        has_game_rule = { rule = ceasefire_break_cooldown_rule option = DAYS_30 }
                        has_country_flag = { flag = ceasefire_break_cooldown days > 29 }
                        ROOT = { has_country_flag = { flag = ceasefire_break_cooldown days > 29 } }
                    }
                    AND = {
                        has_game_rule = { rule = ceasefire_break_cooldown_rule option = DAYS_60 }
                        has_country_flag = { flag = ceasefire_break_cooldown days > 59 }
                        ROOT = { has_country_flag = { flag = ceasefire_break_cooldown days > 59 } }
                    }
                    AND = {
                        has_game_rule = { rule = ceasefire_break_cooldown_rule option = DAYS_180 }
                        has_country_flag = { flag = ceasefire_break_cooldown days > 179 }
                        ROOT = { has_country_flag = { flag = ceasefire_break_cooldown days > 179 } }
                    }
                    AND = {
                        NOT = { has_game_rule = { rule = ceasefire_break_cooldown_rule option = DAYS_30 } }
                        NOT = { has_game_rule = { rule = ceasefire_break_cooldown_rule option = DAYS_60 } }
                        NOT = { has_game_rule = { rule = ceasefire_break_cooldown_rule option = DAYS_180 } }
                        has_country_flag = { flag = ceasefire_break_cooldown days > 89 }
                        ROOT = { has_country_flag = { flag = ceasefire_break_cooldown days > 89 } }
                    }
                }
            }
        }

        send_description = ceasefire_break_send_desc

        allowed = { always = yes }

         complete_effect = {
             # Context: ROOT = sender (breaker), current scope = receiver (target)
             log = "[CeasefireMechanics] Breaking ceasefire - Breaker: [ROOT.GetName], Target: [This.GetName]"

             # Build temporary camps for this break action.
             # - ceasefire_tmp_breaker_camp: countries that should join ROOT in the resumed war
             # - ceasefire_tmp_target_camp: countries that should join PREV in the resumed war
             every_country = { clr_country_flag = ceasefire_tmp_breaker_camp }
             every_country = { clr_country_flag = ceasefire_tmp_target_camp }
 
              # Build temporary camps for this ceasefire pair.
              # Start with legacy storage (AGAINST flags), then (if available) refine using WITH flags to avoid mixing multiple ceasefires.
              # - Countries on ROOT's side were stored as "against PREV"
              # - Countries on PREV's side were stored as "against ROOT"
              every_country = {
                  limit = { has_country_flag = ceasefire_participant_against_@PREV }
                  set_country_flag = ceasefire_tmp_breaker_camp
              }
              every_country = {
                  limit = { has_country_flag = ceasefire_participant_against_@ROOT }
                  set_country_flag = ceasefire_tmp_target_camp
              }
              # Refine camps using WITH flags (new saves) if at least one matching participant exists for this pair.
              if = {
                  limit = {
                      any_country = {
                          has_country_flag = ceasefire_participant_with_@ROOT
                          has_country_flag = ceasefire_participant_against_@PREV
                      }
                  }
                  every_country = {
                      limit = {
                          has_country_flag = ceasefire_tmp_breaker_camp
                          NOT = { has_country_flag = ceasefire_participant_with_@ROOT }
                      }
                      clr_country_flag = ceasefire_tmp_breaker_camp
                  }
              }
              if = {
                  limit = {
                      any_country = {
                          has_country_flag = ceasefire_participant_with_@PREV
                          has_country_flag = ceasefire_participant_against_@ROOT
                      }
                  }
                  every_country = {
                      limit = {
                          has_country_flag = ceasefire_tmp_target_camp
                          NOT = { has_country_flag = ceasefire_participant_with_@PREV }
                      }
                      clr_country_flag = ceasefire_tmp_target_camp
                  }
              }

              # Ensure signatories are part of the temporary camps (for truce/NAP/wargoal cleanup), even in save-compat fallback cases.
              ROOT = { set_country_flag = ceasefire_tmp_breaker_camp }
              set_country_flag = ceasefire_tmp_target_camp

              # STEP 1: Remove ceasefire NAP + truce for all stored participants (both sides, full mesh)
              every_country = {
                 limit = { has_country_flag = ceasefire_tmp_breaker_camp }
                 every_country = {
                     limit = { has_country_flag = ceasefire_tmp_target_camp }
                    diplomatic_relation = { country = PREV relation = non_aggression_pact active = no }
                    set_truce = { target = PREV days = 0 }
                    PREV = {
                        diplomatic_relation = { country = PREV relation = non_aggression_pact active = no }
                        set_truce = { target = PREV days = 0 }
                    }

                    # Remove all ceasefire-created wargoals now that hostilities have resumed.
                    remove_wargoal = { type = ceasefire_resume_war target = PREV }
                    PREV = { remove_wargoal = { type = ceasefire_resume_war target = PREV } }
                }
             }

             # Opinion impact: target side considers the ceasefire broken (apply to breaker country).
             every_country = {
                limit = { has_country_flag = ceasefire_tmp_target_camp }
                add_opinion_modifier = { target = ROOT modifier = broke_ceasefire }
             }

            # Events (breaker/target + player-controlled participants), gated by game rule.
            if = {
                limit = { has_game_rule = { rule = ceasefire_events_rule option = ENABLED } }

                # Breaker perspective (ROOT)
                ROOT = {
                    if = {
                        limit = { is_ai = no }
                        random_list = {
                            20 = { country_event = { id = ceasefire_events.30 } }
                            20 = { country_event = { id = ceasefire_events.31 } }
                            20 = { country_event = { id = ceasefire_events.32 } }
                            20 = { country_event = { id = ceasefire_events.33 } }
                            20 = { country_event = { id = ceasefire_events.34 } }
                        }
                    }
                }

                # Target perspective (PREV / current scope)
                if = {
                    limit = { is_ai = no }
                    ROOT = {
                        PREV = {
                            random_list = {
                                20 = { country_event = { id = ceasefire_events.35 } }
                                20 = { country_event = { id = ceasefire_events.36 } }
                                20 = { country_event = { id = ceasefire_events.37 } }
                                20 = { country_event = { id = ceasefire_events.38 } }
                                20 = { country_event = { id = ceasefire_events.39 } }
                            }
                        }
                    }
                }

                 # Participant perspectives
                 # Breaker camp participants: trigger from target so FROM is target.
                 every_country = {
                     limit = {
                         is_ai = no
                        has_country_flag = ceasefire_tmp_breaker_camp
                         NOT = { tag = ROOT }
                         NOT = { tag = PREV }
                     }
                     random_list = {
                         34 = { country_event = { id = ceasefire_events.30 } }
                        33 = { country_event = { id = ceasefire_events.33 } }
                        33 = { country_event = { id = ceasefire_events.34 } }
                    }
                }
                 # Victim camp participants: trigger from breaker so FROM is breaker.
                 ROOT = {
                     every_country = {
                         limit = {
                             is_ai = no
                            has_country_flag = ceasefire_tmp_target_camp
                             NOT = { tag = ROOT }
                             NOT = { tag = PREV }
                         }
                         random_list = {
                             34 = { country_event = { id = ceasefire_events.35 } }
                            33 = { country_event = { id = ceasefire_events.38 } }
                            33 = { country_event = { id = ceasefire_events.39 } }
                        }
                    }
                }
            }

            # STEP 2: Clear ceasefire tracking flags for the leader pair
            clr_country_flag = ceasefire_active_with_@ROOT
            clr_country_flag = ceasefire_initiated_by_@ROOT
            clr_country_flag = ceasefire_initiator_with_@ROOT
            clr_country_flag = ceasefire_honored_notified_with_@ROOT
            ROOT = { clr_country_flag = ceasefire_active_with_@PREV }
            ROOT = { clr_country_flag = ceasefire_initiated_by_@PREV }
            ROOT = { clr_country_flag = ceasefire_initiator_with_@PREV }
            ROOT = { clr_country_flag = ceasefire_honored_notified_with_@PREV }

            # STEP 3: Re-declare war between the leaders
            # Mark in-progress so on_actions doesn't wipe participant flags before we can add countries back to the war.
            set_country_flag = ceasefire_break_in_progress
            ROOT = { set_country_flag = ceasefire_break_in_progress }
            ROOT = {
                declare_war_on = {
                    target = PREV
                    type = ceasefire_resume_war
                }
            }

            # EXPERIMENTAL (Escrow Ownership): restore original owners while keeping occupation controllers.
            # This makes resumed wars + peace conferences behave as if ownership never permanently changed.
            if = {
                limit = { has_game_rule = { rule = ceasefire_territory_rule option = ESCROW_OWNERSHIP } }
                every_state = {
                    limit = {
                        has_state_flag = ceasefire_escrow_state
                        has_variable = ceasefire_escrow_original_owner
                        has_variable = ceasefire_transfer_to
                        OR = {
                            AND = {
                                var:ceasefire_transfer_to = { has_country_flag = ceasefire_tmp_breaker_camp }
                                var:ceasefire_escrow_original_owner = { has_country_flag = ceasefire_tmp_target_camp }
                            }
                            AND = {
                                var:ceasefire_transfer_to = { has_country_flag = ceasefire_tmp_target_camp }
                                var:ceasefire_escrow_original_owner = { has_country_flag = ceasefire_tmp_breaker_camp }
                            }
                        }
                    }
                    var:ceasefire_escrow_original_owner = { set_state_owner = PREV }
                    var:ceasefire_transfer_to = { set_state_controller = PREV }
                    clr_state_flag = ceasefire_escrow_state
                    clear_variable = ceasefire_escrow_original_owner
                    clear_variable = ceasefire_transfer_to
                }
            }

             # Mark stored participants in-progress too so on_actions won't run the participant-break handler while restoring the war.
             every_country = {
                limit = { has_country_flag = ceasefire_tmp_breaker_camp }
                set_country_flag = ceasefire_break_in_progress
             }
             every_country = {
                limit = { has_country_flag = ceasefire_tmp_target_camp }
                set_country_flag = ceasefire_break_in_progress
             }

             # STEP 4: Add all stored participants back into the same war
             every_country = {
                 limit = {
                    has_country_flag = ceasefire_tmp_breaker_camp
                    NOT = { tag = ROOT }
                    NOT = { has_war_with = PREV }
                    NOT = { is_in_faction_with = PREV }
                 }
                 add_to_war = {
                     targeted_alliance = ROOT
                     enemy = PREV
                }
             }
              every_country = {
                  limit = {
                     has_country_flag = ceasefire_tmp_target_camp
                     NOT = { tag = PREV }
                     NOT = { has_war_with = ROOT }
                     NOT = { is_in_faction_with = ROOT }
                  }
                  add_to_war = {
                      targeted_alliance = PREV
                      enemy = ROOT
                 }
            }

             # Clear in-progress marker for participants before wiping participant flags.
             every_country = {
                limit = { has_country_flag = ceasefire_tmp_breaker_camp }
                clr_country_flag = ceasefire_break_in_progress
             }
             every_country = {
                limit = { has_country_flag = ceasefire_tmp_target_camp }
                clr_country_flag = ceasefire_break_in_progress
             }

             # STEP 5: Cleanup participant flags (this ceasefire pair only)
             every_country = {
                limit = { has_country_flag = ceasefire_tmp_breaker_camp }
                clr_country_flag = ceasefire_participant_against_@PREV
             }
             every_country = {
                limit = { has_country_flag = ceasefire_tmp_target_camp }
                clr_country_flag = ceasefire_participant_against_@ROOT
             }
 
              # Clear temporary camps used during restore.
              every_country = { clr_country_flag = ceasefire_tmp_breaker_camp }
              every_country = { clr_country_flag = ceasefire_tmp_target_camp }

            # STEP 6: Clear in-progress marker
            clr_country_flag = ceasefire_break_in_progress
            ROOT = { clr_country_flag = ceasefire_break_in_progress }

            # Cleanup ceasefire_active if no other ceasefires remain
            ROOT = {
                if = {
                    limit = {
                        NOT = {
                            any_country = {
                                PREV = { has_country_flag = ceasefire_active_with_@PREV }
                            }
                        }
                    }
                    clr_country_flag = ceasefire_active
                }
            }
            if = {
                limit = {
                    NOT = {
                        any_country = {
                            PREV = { has_country_flag = ceasefire_active_with_@PREV }
                        }
                    }
                }
                clr_country_flag = ceasefire_active
            }

            log = "[CeasefireMechanics] Ceasefire broken - war restored between camps of [ROOT.GetName] and [This.GetName]"
        }

        ai_desire = {
            # current scope = target (receiver), ROOT = sender (AI deciding to break)
            # Breaking a ceasefire should be rare and opportunistic.
            base = -170

            # Game rule: optionally block AI from breaking ceasefires entirely.
            modifier = {
                add = -10000
                has_game_rule = { rule = ceasefire_ai_break_rule option = BLOCKED }
            }

            # Don't break if we're in trouble (in any ongoing wars).
            modifier = { add = -240 ROOT = { surrender_progress > 0.50 } }
            modifier = { add = -140 ROOT = { has_war_support < 0.30 } }
            modifier = { add = -90 ROOT = { has_stability < 0.35 } }
            modifier = { add = -60 ROOT = { has_manpower < 80000 } }
            modifier = { add = -90 ROOT = { has_manpower < 30000 } }

            # Break when the target is vulnerable (usually because they're losing another war).
            modifier = { add = 40 surrender_progress > 0.35 }
            modifier = { add = 90 surrender_progress > 0.55 }
            modifier = { add = 150 surrender_progress > 0.75 }

            # Relative strength vs the target (from the target's perspective).
            # If ROOT is much stronger than THIS, breaking becomes more tempting.
            modifier = {
                set_temp_variable = { rel = THIS.enemies_strength_ratio@ROOT }
                multiply_temp_variable = { rel = 220 } # ratio->points
                round_temp_variable = rel
                subtract_from_temp_variable = { rel = 220 } # ratio=1.0 -> 0
                clamp_temp_variable = { var = rel min = -160 max = 240 }
                add = var:rel
            }

            # Targets that are majors are riskier to re-engage unless they're already collapsing.
            modifier = { add = -35 is_major = yes surrender_progress < 0.20 }

            # If we have a wargoal ready, more willing to resume hostilities.
            modifier = {
                add = 55
                ROOT = { has_wargoal_against = PREV }
            }

            # Political signal: democracies are less inclined to break truces without a strong advantage.
            modifier = { add = -20 ROOT = { has_government = democratic } }
            modifier = { add = 10 ROOT = { has_government = communism } }
            modifier = { add = 15 ROOT = { has_government = fascism } }

            # Majors are slightly more willing to resume wars once positioned well.
            modifier = { add = 10 ROOT = { is_major = yes } }
        }
    }
}
