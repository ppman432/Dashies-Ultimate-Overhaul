scripted_gui = {
	
	state_interactions_menu_gui = {

		context_type = selected_state_context

		window_name = "state_interaction_menu_container"

		parent_window_name = "countrystateview"

		visible = {
			always = yes
		}

		effects = {
			
			state_interaction_menu_button_click = {
				if = {
					limit = {
						ROOT = { has_country_flag = SI_menu_flag }
					}
					ROOT = {
						clr_country_flag = SI_menu_flag
					}
				}
				else = {
					ROOT = {
						set_country_flag = SI_menu_flag
					}
				}
			}

		}
	}


	######  ##  ##  ######  ######  ####    ######  ######  ######  ######  ######  ##  ##  ######
	  ##    ### ##    ##    ##      ##  ##  ##  ##  ##        ##      ##    ##  ##  ### ##  ##

	state_interactions_menuview_gui = {

		context_type = selected_state_context

		window_name = "state_interaction_container"

		parent_window_name = "countrystateview"

		visible = {
			ROOT = {
				OR = {
					is_ai = yes
					has_country_flag = SI_menu_flag
				}
			}
		}

		properties = {
			interaction_debug_button = {
				frame = global.SI_debug_var
			}
		}

		######  ######  ######  ######  ######  ######  ######
		##      ##      ##      ##      ##        ##    ##
		######  ######  ######  ######  ##        ##    ######
		##      ##      ##      ##      ##        ##        ##
		######  ##      ##      ######  ######    ##    ######

		effects = {
			interaction_debug_button_click = {
				if = {
					limit = {
						check_variable = { global.SI_debug_var = 1 }
					}
					set_variable = { global.SI_debug_var = 2 }
					set_global_flag = SI_debug_flag
				}
				else = {
					set_variable = { global.SI_debug_var = 1 }
					clr_global_flag = SI_debug_flag
				}
			}

			state_menu_transfer_menu_button_click = {
				ROOT = {
					if = {
						limit = { NOT = { has_country_flag = transfer_enabled } }
						set_country_flag = transfer_enabled
					}
					else = {
						clr_country_flag = transfer_enabled
					}
				}
			}

			state_menu_demand_state_button_click = {
				set_variable = { ROOT.requested_state_@FROM.owner = FROM }

				set_variable = { ROOT.request_target_civilian_factories_@THIS = var:FROM.industrial_complex_level }
				multiply_variable = { ROOT.request_target_civilian_factories_@THIS = 5 }

				set_variable = { ROOT.request_target_military_factories_@THIS = var:FROM.arms_factory_level }
				multiply_variable = { ROOT.request_target_military_factories_@THIS = 10 }

				FROM.owner = {
					set_variable = { ROOT.state_factor_var_@THIS = ROOT.owned_states }
					divide_variable = { ROOT.state_factor_var_@THIS = THIS.owned_states }
					clamp_variable = {
						var = ROOT.state_factor_var_@THIS
						min = 0.25
						max = 2.0
					}

					set_variable = { ROOT.relations_value_positive_scaled_@THIS = THIS.opinion@ROOT }
					clamp_variable = { var = ROOT.relations_value_positive_scaled_@THIS min = 0 }
					divide_variable = { ROOT.relations_value_positive_scaled_@THIS = 3 }
					round_variable = ROOT.relations_value_positive_scaled_@THIS
					# ^^^ At 100 opinion youll get about 33 points or 0 if original value is negative.
	
					set_variable = { ROOT.relations_value_negative_scaled_@THIS = THIS.opinion@ROOT }
					multiply_variable = { ROOT.relations_value_negative_scaled_@THIS = -1 }
					divide_variable = { ROOT.relations_value_negative_scaled_@THIS = 3 }
					round_variable = ROOT.relations_value_negative_scaled_@THIS
					multiply_variable = { ROOT.relations_value_negative_scaled_@THIS = -1 }
					clamp_variable = { var = ROOT.relations_value_negative_scaled_@THIS max = 0 }
					# ^^^ At -100 opinion youll get about -33 points or 0 if original value is positive.

					if = {
						limit = {
							NOT = { has_idea = SI_AI_reluctance_idea } # So that the reluctance affects second and onwards demands
						}
						set_variable = { THIS.si_ai_reluctance_stage = -1 }
					}
					add_to_variable = { THIS.si_ai_reluctance_stage = 1 }
					set_variable = { THIS.si_ai_reluctance_value = global.si_ai_reluctance_points_per_stage }
					multiply_variable = { THIS.si_ai_reluctance_value = THIS.si_ai_reluctance_stage }
					multiply_variable = { THIS.si_ai_reluctance_value = -1 }

					add_timed_idea = {
						idea = SI_AI_reluctance_idea
						days = var:global.si_ai_reluctance_time_days
					}

					country_event = {
						id = sievent.2
						days = 1
						random_days = 6
					}
				}
				FROM = {
					set_state_flag = {
						flag = SI_state_demand_subject_flag
						days = 21
					}
				}
				ROOT = {
					add_political_power = var:global.si_demand_state_pp_take
					add_timed_idea = {
						idea = SI_major_demands_idea
						days = var:global.si_demand_state_cooldown_days
					}

					if = {
						limit = {
							ROOT = { is_major = no }
							FROM.owner = { is_major = no }
						}
						add_named_threat = {
							threat = 0.5
							name = SI_state_demanded_threat
						}
					}
					else_if = {
						limit = {
							OR = {
								AND = {
									ROOT = { is_major = yes }
									FROM.owner = { is_major = no }
								}
								AND = {
									ROOT = { is_major = no }
									FROM.owner = { is_major = yes }
								}
							}
						}
						add_named_threat = {
							threat = 1
							name = SI_state_demanded_threat
						}
					}
					else = {
						add_named_threat = {
							threat = 1.5
							name = SI_state_demanded_threat
						}
					}
				}

				# Debug
				if = {
					limit = {
						ROOT = { is_ai = yes }
						is_debug = yes
					}
					log = "(SI) [ROOT.GetName] demanded [FROM.GetName] from [FROM.owner.GetName]"
				}
			}

			state_menu_demand_vsl_button_click = {
				FROM.owner = {
					set_variable = { ROOT.state_factor_var_@THIS = ROOT.owned_states }
					divide_variable = { ROOT.state_factor_var_@THIS = THIS.owned_states }
					clamp_variable = {
						var = ROOT.state_factor_var_@THIS
						min = 0.25
						max = 2.0
					}

					set_variable = { ROOT.relations_value_positive_scaled_@THIS = THIS.opinion@ROOT }
					clamp_variable = { var = ROOT.relations_value_positive_scaled_@THIS min = 0 }
					divide_variable = { ROOT.relations_value_positive_scaled_@THIS = 3 }
					round_variable = ROOT.relations_value_positive_scaled_@THIS
					# ^^^ At 100 opinion youll get about 33 points or 0 if original value is negative.
	
					set_variable = { ROOT.relations_value_negative_scaled_@THIS = THIS.opinion@ROOT }
					multiply_variable = { ROOT.relations_value_negative_scaled_@THIS = -1 }
					divide_variable = { ROOT.relations_value_negative_scaled_@THIS = 3 }
					round_variable = ROOT.relations_value_negative_scaled_@THIS
					multiply_variable = { ROOT.relations_value_negative_scaled_@THIS = -1 }
					clamp_variable = { var = ROOT.relations_value_negative_scaled_@THIS max = 0 }
					# ^^^ At -100 opinion youll get about -33 points or 0 if original value is positive.

					if = {
						limit = {
							NOT = { has_idea = SI_AI_reluctance_idea } # So that the reluctance affects second and onwards demands
						}
						set_variable = { THIS.si_ai_reluctance_stage = -1.5 }
					}
					add_to_variable = { THIS.si_ai_reluctance_stage = 1.5 }
					set_variable = { THIS.si_ai_reluctance_value = global.si_ai_reluctance_points_per_stage }
					multiply_variable = { THIS.si_ai_reluctance_value = THIS.si_ai_reluctance_stage }
					multiply_variable = { THIS.si_ai_reluctance_value = -1 }

					add_timed_idea = {
						idea = SI_AI_reluctance_idea
						days = var:global.si_ai_reluctance_time_days
					}

					set_country_flag = {
						flag = SI_country_demand_subject_flag
						days = 21
					}
					country_event = {
						id = sievent.4
						days = 1
						random_days = 6
					}
				}
				ROOT = {
					add_political_power = var:global.si_demand_vassalisation_pp_take
					add_timed_idea = {
						idea = SI_major_demands_idea
						days = var:global.si_demand_vassalisation_cooldown_days
					}
					ROOT = {
						if = {
							limit = {
								ROOT = { is_major = no }
								FROM.owner = { is_major = no }
							}
							add_named_threat = {
								threat = 1
								name = SI_vassalisation_demanded_threat
							}
						}
						else_if = {
							limit = {
								OR = {
									AND = {
										ROOT = { is_major = yes }
										FROM.owner = { is_major = no }
									}
									AND = {
										ROOT = { is_major = no }
										FROM.owner = { is_major = yes }
									}
								}
							}
							add_named_threat = {
								threat = 2
								name = SI_vassalisation_demanded_threat
							}
						}
						else = {
							add_named_threat = {
								threat = 3
								name = SI_vassalisation_demanded_threat
							}
						}
					}
				}

				# Debug
				if = {
					limit = {
						ROOT = { is_ai = yes }
						is_debug = yes
					}
					log = "(SI) [ROOT.GetName] demanded [FROM.owner.GetName] vassalisation"
				}
			}

			

			check_core_button_click = {
				every_country = { every_state = { clr_state_flag = core_cost_checked } }
				ROOT = {
					set_variable = { cost_to_core = PREV.state_population_k }
					CWC++_cost_divider = yes
					round_variable = cost_to_core

					set_variable = { cost_to_claim = cost_to_core }
					divide_variable = { cost_to_claim = 2 }
				}
				PREV = { set_state_flag = core_cost_checked }
			}

			core_state_button_click = {
				log = [?cost_to_core]
				ROOT = {
					multiply_variable = { cost_to_core = -1 }
					add_political_power = cost_to_core
					add_state_core = PREV
					multiply_variable = { cost_to_core = -1 }
				}
			}

			claim_state_button_click = {
				ROOT = {
					multiply_variable = { cost_to_claim = -1 }
					add_political_power = cost_to_claim 
					add_state_claim = PREV
					multiply_variable = { cost_to_claim = -1 }
				}
			}

			revoke_core_button_click = {
				ROOT = {
					set_variable = { landlord_state = PREV }
					set_country_flag = landlord_revoking_core
					clr_country_flag = landlord_returning_state
				}
			}

			revoke_claim_button_click = {
				ROOT = {
					set_variable = { landlord_state = PREV }
					set_country_flag = landlord_revoking_claim
					clr_country_flag = landlord_returning_state
				}
			}

			state_menu_move_capital_button_click = {
				ROOT = {
					set_capital = { state = PREV }
				}

				if = {
					limit = {
						NOT = {
							has_global_flag = SI_debug_flag
						}
					}
					ROOT = {
						add_political_power = var:global.si_move_capital_pp_take
						add_timed_idea = {
							idea = SI_else_demands_idea
							days = var:global.si_move_capital_cooldown_days
						}
					}
				}
			}
		}

		######  ####    ######  ######  ######  ######  ####    ######
          ##    ##  ##    ##    ##      ##      ##      ##  ##  ##
          ##    ####      ##    ##  ##  ##  ##  ######  ####    ######
          ##    ##  ##    ##    ##  ##  ##  ##  ##      ##  ##      ##
          ##    ##  ##  ######  ######  ######  ######  ##  ##  ######

		triggers = {
			interaction_debug_button_visible = {
				is_debug = yes
			}
			check_core_button_click_enabled = {
				NOT = {
					PREV = { has_state_flag = core_cost_checked }
				}
			}

			core_state_button_click_enabled = {
				PREV = { has_state_flag = core_cost_checked }

				OR  = {
					check_variable = { ROOT.cost_to_core < ROOT.political_power }
					check_variable = { ROOT.cost_to_core = ROOT.political_power }
				}

				PREV = {
					NOT = { is_core_of = ROOT }

					# Occupation check
					if = {
						limit = {
							has_game_rule = {
								rule = CWC_occupy_to_core
								option = CWC_OCCUPY_PEACE
							}
						}
						is_owned_and_controlled_by = ROOT
					}
					if = {
						limit = {
							has_game_rule = {
								rule = CWC_occupy_to_core
								option = CWC_OCCUPY_WAR
							}
						}
						ROOT = { has_full_control_of_state = PREV }
					}


					# Compliance check
					if = {
						limit = {
							has_game_rule = {
								rule = CWC_comp_to_core
								option = CWC_COMP_COST_100
							}
						}
						compliance > 99.99
					}
					if = {
						limit = {
							has_game_rule = {
								rule = CWC_comp_to_core
								option = CWC_COMP_COST_80
							}
						}
						compliance > 79.99
					}
					else_if = {
						limit = {
							has_game_rule = {
								rule = CWC_comp_to_core
								option = CWC_COMP_COST_60
							}
						}
						compliance > 59.99
					}
					else_if = {
						limit = {
							has_game_rule = {
								rule = CWC_comp_to_core
								option = CWC_COMP_COST_40
							}
						}
						compliance > 29.99
					}
					else_if = {
						limit = {
							has_game_rule = {
								rule = CWC_comp_to_core
								option = CWC_COMP_COST_20
							}
						}
						compliance > 19.99
					}
				}
			}

			claim_state_button_click_enabled = {
				PREV = {
					has_state_flag = core_cost_checked
					NOT = { is_core_of = ROOT }
					NOT = { is_claimed_by = ROOT }
				}
				NOT = { ROOT = { has_full_control_of_state = PREV } }

				OR  = {
					check_variable = { ROOT.cost_to_claim < ROOT.political_power }
					check_variable = { ROOT.cost_to_claim = ROOT.political_power }
				}
			}

			revoke_core_button_click_enabled = {
				PREV = {
					is_core_of = ROOT
					NOT = { is_capital = yes }
				}
			}

			revoke_claim_button_click_enabled = {
				PREV = {
					is_claimed_by = ROOT
				}
			}

			state_menu_transfer_menu_button_visible = {
				ROOT = {
					has_country_flag = ST_HOST
				}
			}

			state_menu_demand_state_button_click_enabled = {
				OR = {
					has_global_flag = SI_debug_flag
					AND = {
						ROOT = {
							has_political_power > var:global.si_demand_state_pp_required
							is_subject = no
							NOT = {
								has_idea = SI_major_demands_idea
							}
						}
						FROM.owner = {
							NOT = {
								tag = ROOT
								has_war_with = ROOT
								is_subject_of = ROOT
								is_in_faction_with = ROOT
							}
							OR = {
								is_neighbor_of = ROOT
								any_neighbor_country = {
									is_subject_of = ROOT
								}
							}
						}
						FROM = {
							any_neighbor_state = {
								owner = {
									OR = {
										is_subject_of = ROOT
										tag = ROOT
									}
								}
							}
							NOT = {
								has_state_flag = SI_state_demand_subject_flag
							}
						}
					}
				}
			}
			state_menu_demand_vsl_button_click_enabled = {
				# ROOT = { is_ai = no }
				OR = {
					has_global_flag = SI_debug_flag
					AND = {
						ROOT = {
							has_political_power > var:global.si_demand_vassalisation_pp_required
							is_subject = no
							NOT = {
								has_idea = SI_major_demands_idea
							}
						}
						FROM.owner = {
							has_war = no
							NOT = {
								tag = ROOT
								is_subject_of = ROOT
								is_in_faction_with = ROOT
								has_country_flag = SI_country_demand_subject_flag
							}
							OR = {
								is_neighbor_of = ROOT
								any_neighbor_country = {
									is_subject_of = ROOT
								}
							}
						}
					}
				}
			}
			state_menu_claim_state_button_click_enabled = {
				ROOT = { is_ai = no }
				OR = {
					has_global_flag = SI_debug_flag
					AND = {
						ROOT = {
							command_power > var:global.si_claim_state_cp_required
							NOT = {
								has_idea = SI_minor_demands_idea
							}
						}
						FROM.owner = {
							NOT = {
								tag = ROOT
								is_in_faction_with = ROOT
								is_subject_of = ROOT
							}
							OR = {
								is_neighbor_of = ROOT
								any_neighbor_country = {
									is_subject_of = ROOT
								}
							}
						}
						FROM = {
							NOT = {
								is_core_of = ROOT
								is_claimed_by = ROOT
							}
						}
					}
				}
			}
			state_menu_core_state_button_click_enabled = {
				ROOT = { is_ai = no }
				OR = {
					has_global_flag = SI_debug_flag
					AND = {
						ROOT = {
							has_equipment = {
								infantry_equipment > var:global.si_core_state_guns_required
							}
							command_power > var:global.si_core_state_cp_required
							NOT = {
								has_idea = SI_minor_demands_idea
							}
						}
						FROM = {
							resistance < var:global.si_core_state_resistance_required
							compliance > var:global.si_core_state_compliance_required
							NOT = {
								is_core_of = ROOT
							}
						}
						FROM.owner = {
							tag = ROOT
						}
					}
				}
			}

			state_menu_move_capital_button_click_enabled = {
				ROOT = { is_ai = no }
				OR = {
					has_global_flag = SI_debug_flag
					AND = {
						FROM = {
							is_owned_by = ROOT
							is_capital = no
							is_core_of = ROOT
						}
						ROOT = {
							has_political_power > var:global.si_move_capital_pp_required
							NOT = {
								has_idea = SI_else_demands_idea
							}
						}
					}
				}
			}
		}
	}

}
