scripted_gui = {

	st_state_manager_powers = {
	
		context_type = selected_state_context
		parent_window_token = selected_state_view
		window_name = "st_state_manager_powers_container"
		
		visible = {
			ROOT = {
				is_ai = no
				has_country_flag = transfer_enabled
				has_country_flag = SI_menu_flag
			}
		}
		
		effects = {
			
			st_state_manager_power_mark_for_transfer_click = {
				FROM.FROM = { 
					set_variable = { ROOT.marked_for_transfer_@THIS = 1 }
				}
			}
			
			st_state_manager_power_unmark_for_transfer_click = {
				FROM.FROM = { 
					set_variable = { ROOT.marked_for_transfer_@THIS = 0 }
				}
			}
		}
		
		triggers = {
		
			st_state_manager_power_mark_for_transfer_click_enabled = {
					NOT = { check_variable = { ROOT.marked_for_transfer_@THIS > 0 } }
					owner = {
						OR = {
							ROOT = {has_country_flag = ST_HOST}
							is_ai = yes
							check_variable = {ROOT = THIS}
						}
					}
			}
			st_state_manager_power_mark_for_transfer_visible = {
				FROM.FROM = {
					NOT = { check_variable = { ROOT.marked_for_transfer_@THIS > 0 } }
				}
			}
			st_state_manager_power_unmark_for_transfer_visible = {
				FROM.FROM = {
					check_variable = { ROOT.marked_for_transfer_@THIS > 0 }
				}
			}
		}
		
	}
	
}
