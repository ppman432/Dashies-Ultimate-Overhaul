scripted_gui = {
  credits_main_gui = {
    window_name = "main_credits_gui_window"
    context_type = player_context

    visible = {
		has_country_flag = credits_gui_opened
    }

    effects = {
		close_button_click = {
			clr_country_flag = credits_gui_opened
		}

		convert_pp_entry_click = { 
			add_to_variable = {
				var = credits_amount
				value = 1
			}
			add_political_power = -50
			clamp_variable = { 
				var = credits_amount
				min = 0
				max = 99
			}
		}
		
		convert_cp_entry_click = { 
			add_to_variable = {
				var = credits_amount
				value = 1
			}
			add_command_power = -50
			clamp_variable = { 
				var = credits_amount
				min = 0
				max = 99
			}
		}		
		
		common_crate_entry_click = { 
			subtract_from_variable = {
				var = credits_amount
				value = 3
			}
			if = {
				limit = { NOT = { has_country_flag = common_crate_gui_opened } }
				set_country_flag = common_crate_gui_opened
			}
		}
		
		rare_crate_entry_click = { 
			subtract_from_variable = {
				var = credits_amount
				value = 9
			}
			if = {
				limit = { NOT = { has_country_flag = rare_crate_gui_opened } }
				set_country_flag = rare_crate_gui_opened
			}
		}

		epic_crate_entry_click = { 
			subtract_from_variable = {
				var = credits_amount
				value = 12
			}
			if = {
				limit = { NOT = { has_country_flag = epic_crate_gui_opened } }
				set_country_flag = epic_crate_gui_opened
			}
		}			
		
		uncommon_crate_entry_click = { 
			subtract_from_variable = {
				var = credits_amount
				value = 6
			}
			if = {
				limit = { NOT = { has_country_flag = uncommon_crate_gui_opened } }
				set_country_flag = uncommon_crate_gui_opened
			}
		}

		legendary_crate_entry_click = { 
			subtract_from_variable = {
				var = credits_amount
				value = 15
			}
			if = {
				limit = { NOT = { has_country_flag = legendary_crate_gui_opened } }
				set_country_flag = legendary_crate_gui_opened
			}
		}		
		
    }
	
	triggers = {
		convert_pp_entry_click_enabled = { 
			has_political_power > 50
		}
		
		convert_cp_entry_click_enabled = { 
			command_power > 50
		}		
		
		common_crate_entry_click_enabled = { 
			check_variable = {
				var = credits_amount
				value = 3
				compare = greater_than_or_equals
			}
			if = {
				limit = { NOT = { has_country_flag = common_crate_gui_opened } }
			}
		}
		
		uncommon_crate_entry_click_enabled = { 
			check_variable = {
				var = credits_amount
				value = 6
				compare = greater_than_or_equals
			}
			if = {
				limit = { NOT = { has_country_flag = common_crate_gui_opened } }
			}
		}

		rare_crate_entry_click_enabled = { 
			check_variable = {
				var = credits_amount
				value = 9
				compare = greater_than_or_equals
			}
			if = {
				limit = { NOT = { has_country_flag = rare_crate_gui_opened } }
			}
		}

		epic_crate_entry_click_enabled = { 
			check_variable = {
				var = credits_amount
				value = 12
				compare = greater_than_or_equals
			}
			if = {
				limit = { NOT = { has_country_flag = epic_crate_gui_opened } }
			}
		}

		legendary_crate_entry_click_enabled = { 
			check_variable = {
				var = credits_amount
				value = 15
				compare = greater_than_or_equals
			}
			if = {
				limit = { NOT = { has_country_flag = legendary_crate_gui_opened } }
			}
		}		
		
	}
  }
}
