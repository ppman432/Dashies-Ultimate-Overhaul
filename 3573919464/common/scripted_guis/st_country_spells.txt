scripted_gui = {
	
	st_country_powers = {
	
		context_type = selected_country_context
		parent_window_token = selected_state_view
		window_name = "st_subject_powers_container"
		
		visible = {
			ROOT = {
				is_ai = no
				has_country_flag = transfer_enabled
				has_country_flag = SI_menu_flag
			}
		}
		
		effects = {
		
			mark_all_states_click = {
				if = {
					limit = {
						OR = {
							ROOT = {has_country_flag = ST_HOST}
							is_ai = yes
							check_variable = {THIS = ROOT}
						}
					}
					every_owned_state = {
						limit = {
							check_variable = { ROOT.marked_for_transfer_@THIS < 1 }
						}
						set_variable = { ROOT.marked_for_transfer_@THIS = 1 }
					}
				}
			}
			
			state_transfer_core_button_click = {
				set_temp_variable = {taker = this}
				log = "[ROOT.Getname] started coring states on [?global.date.GetDateString]."
				every_state = {
					limit = {
						check_variable = { ROOT.marked_for_transfer_@THIS > 0 }
						owner = {
							OR = {
								ROOT = {has_country_flag = ST_HOST}
								is_ai = yes
								check_variable = {THIS = ROOT}
							}
						}
					}
					log = "coring State [THIS.GetName] which is owned by [THIS.owner.GetName] to [?taker.GetName]."
					add_core_of = taker
					set_variable = {ROOT.marked_for_transfer_@THIS = 0 }
				}
				log = "Ending Transfer."
			}
			
		
			mark_colonial_states_click  = {
				if = {
					limit = {
						OR = {
							ROOT = {has_country_flag = ST_HOST}
							is_ai = yes
							check_variable = {THIS = ROOT}
						}
					}
					every_owned_state = {
						limit = {
							NOT = { is_in_home_area = yes }
							NOT = { is_core_of = PREV }
							check_variable = { ROOT.marked_for_transfer_@THIS < 1 }
						}
						set_variable = { ROOT.marked_for_transfer_@THIS = 1 }
					}
				}
			}
		
			clear_all_markings_click = {
				every_state = {
					set_variable = {ROOT.marked_for_transfer_@THIS = 0}
				}
			}
			
			award_states_click = {
				set_temp_variable = {taker = this}
				log = "[ROOT.Getname] started transferring states on [?global.date.GetDateString]."
				every_state = {
					limit = {
						check_variable = { ROOT.marked_for_transfer_@THIS > 0 }
						owner = {
							OR = {
								ROOT = {has_country_flag = ST_HOST}
								is_ai = yes
								check_variable = {THIS = ROOT}
							}
						}
					}
					log = "Transfering State [THIS.GetName] which is owned by [THIS.owner.GetName] to [?taker.GetName]."
					var:taker = { transfer_state = PREV }
					set_variable = {ROOT.marked_for_transfer_@THIS = 0 }
				}
				log = "Ending Transfer."
			}
			
		#	core_states_click = {

		#		}

		}
		
		triggers = {
		
			mark_all_states_click_enabled = {
				OR = {
					ROOT = {has_country_flag = ST_HOST}
					is_ai = yes
					check_variable = {THIS = ROOT}
				}
				any_owned_state = {
					check_variable = { ROOT.marked_for_transfer_@THIS < 1 }
				}
			}
		
			mark_colonial_states_click_enabled = {
				OR = {
					ROOT = {has_country_flag = ST_HOST}
					is_ai = yes
					check_variable = {THIS = ROOT}
				}
				any_owned_state = {
					NOT = { is_in_home_area = yes }
					check_variable = { ROOT.marked_for_transfer_@THIS < 1 }
					NOT = { is_core_of = PREV }
				}
			}
			
			clear_all_markings_click_enabled = {
				any_state = {
					check_variable = { ROOT.marked_for_transfer_@THIS > 0 }
				}
			}
		
			award_states_click_enabled = {
				set_temp_variable = {taker = THIS}
				
				any_state = {
					owner = {
						OR = {
							ROOT = {has_country_flag = ST_HOST}
							is_ai = yes
							check_variable = {THIS = ROOT}
						}
					}
					check_variable = { ROOT.marked_for_transfer_@THIS > 0 }
				}
				NOT = {
					any_state = {
						is_owned_by = taker
						check_variable = { ROOT.marked_for_transfer_@THIS > 0 }
					}
				}
			}
		
		}
        
	}
