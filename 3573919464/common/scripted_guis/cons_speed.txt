scripted_gui = {
	construction_speed_view_opener_menu = {
		context_type = player_context
		
		window_name = "cons_speed_opener_container"
		parent_window_token = construction_tab

		visible = {
			always = yes
		}
		
		ai_enabled = {
			always = no
		}
		
		effects = {
			construction_speed_open_click = {
				if = {
					limit = { NOT = { has_country_flag = cons_speed_on } }
					set_country_flag = cons_speed_on
				}
				else = { clr_country_flag = cons_speed_on }
			}
		}
	}
	
	construction_speed_panel_window = {
		context_type = player_context

		window_name = "construction_speed_panel_window"
		parent_window_token = construction_tab

		visible = {
			always = yes
		}

		ai_enabled = {
			always = no
		}
	}
	
	construction_speed_view_menu = {
		context_type = player_context
		
		window_name = "cons_speed_container"
		parent_window_name = "construction_speed_panel_window_instance"

		visible = {
			NOT = { has_country_flag = cons_speed_on }
		}
		
		triggers = {
			unlockable_constructions_visible = {
				OR = {
					has_tech = commercial_nuclear_reactor_tech
					has_tech = stronghold_network_tech
					has_tech = abm_system_tech
					has_tech = land_multi_charge_large_caliber_gun_tech
					has_tech = hardened_missile_silo_tech
					has_tech = nuclear_reactor
					has_tech = better_reactors
				}
			}
			unlockable_title_visible = {
				OR = {
					has_tech = commercial_nuclear_reactor_tech
					has_tech = stronghold_network_tech
					has_tech = abm_system_tech
					has_tech = land_multi_charge_large_caliber_gun_tech
					has_tech = hardened_missile_silo_tech
					has_tech = nuclear_reactor
					has_tech = better_reactors
				}
			}
			commercial_nuclear_reactor_speed_visible = {
				has_tech = commercial_nuclear_reactor_tech
			}
			stronghold_network_speed_visible = {
				has_tech = stronghold_network_tech
			}
			anti_ballistic_speed_visible = {
				has_tech = abm_system_tech
			}
			caliber_gun_speed_visible = {
				has_tech = land_multi_charge_large_caliber_gun_tech
			}
			hardened_silo_speed_visible = {
				has_tech = hardened_missile_silo_tech
			}
			nuclear_reactor_speed_visible = {
				has_tech = nuclear_reactor
			}
			modern_nuclear_reactor_speed_visible = {
				has_tech = better_reactors
			}
		}
		
		ai_enabled = {
			always = no
		}
	}
}
